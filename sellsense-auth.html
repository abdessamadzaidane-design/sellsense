// ---------- STATE ---------- let lang = 'fr'; let selType = null; // ---------- UI HELPERS ---------- function clr(){ document.getElementById('ae').classList.remove('show'); document.getElementById('as').classList.remove('show'); } function err(m){ const e=document.getElementById('ae'); e.textContent=m; e.classList.add('show'); } function okMsg(m){ const s=document.getElementById('as'); s.textContent=m; s.classList.add('show'); } function ld(id,on){ const b=document.getElementById(id); b.classList.toggle('ld',on); b.disabled=on; } // ---------- LOGIN ---------- async function doLogin(){ clr(); const email=document.getElementById('le').value.trim(); const pass=document.getElementById('lp').value; if(!email||!pass){ err('Remplis tous les champs.'); return; } ld('bl',true); const {error}=await sb.auth.signInWithPassword({ email, password:pass }); ld('bl',false); if(error){ err(error.message); return; } window.location.href='sellsense-dashboard.html'; } // ---------- SIGNUP ---------- async function doSignup(){ clr(); const name=document.getElementById('sn').value.trim(); const email=document.getElementById('se').value.trim(); const pass=document.getElementById('sp').value; if(!name){ err('Entre ton nom.'); return; } if(!email){ err('Entre ton e-mail.'); return; } if(pass.length<8){ err('Mot de passe trop court (min. 8).'); return; } if(!selType){ err('Choisis ton type d’activité.'); return; } ld('bs',true); const {data,error}=await sb.auth.signUp({ email, password:pass, options:{ data:{ full_name:name, business_type:selType } } }); ld('bs',false); if(error){ err(error.message); return; } // Safe profile creation if(data.user){ try{ await sb.from('profiles').upsert({ id:data.user.id, full_name:name, email, business_type:selType, currency:'USD' }); }catch(e){ console.log('Profile skipped'); } } okMsg('✓ Compte créé ! Vérifie ton e-mail.'); } // ---------- GOOGLE ---------- async function gLogin(){ const {error}=await sb.auth.signInWithOAuth({ provider:'google', options:{ redirectTo:window.location.origin+'/sellsense-dashboard.html' } }); if(error){ err(error.message); } } // ---------- RESET ---------- async function doReset(){ clr(); const email=document.getElementById('re').value.trim(); if(!email){ err('Entre ton e-mail.'); return; } ld('br',true); const {error}=await sb.auth.resetPasswordForEmail(email,{ redirectTo:window.location.origin+'/sellsense-auth.html' }); ld('br',false); if(error){ err(error.message); return; } okMsg('✓ Email envoyé !'); } // ---------- TYPE SELECT ---------- function selT(t,el){ selType=t; document.querySelectorAll('.tc').forEach(c=>c.classList.remove('sel')); el.classList.add('sel'); } // ---------- ENTER KEY ---------- document.addEventListener('keydown',e=>{ if(e.key==='Enter'){ if(document.getElementById('vlogin').style.display!=='none'){ doLogin(); }else if(document.getElementById('vsignup').style.display!=='none'){ doSignup(); } } });
